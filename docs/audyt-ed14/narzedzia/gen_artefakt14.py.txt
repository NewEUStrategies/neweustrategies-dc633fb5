# ---------- markdown -> HTML ----------
def inline(t, pogrubienia=True):
    t = html.escape(t.strip(), quote=False)
    kody = []

    def schowaj(m):
        kody.append('<code>' + m.group(1) + '</code>')
        return f'\x01{len(kody) - 1}\x02'
    t = re.sub(r'`([^`]+)`', schowaj, t)
    if pogrubienia:
        t = re.sub(r'\*\*(.+?)\*\*', r'<b>\1</b>', t)
    return re.sub(r'\x01(\d+)\x02', lambda m: kody[int(m.group(1))], t)


def sekcja(nr):
    m = re.search(r'^### 18\.%d\. .*$' % nr, rozdzial, re.M)
    n = re.search(r'^### 18\.%d\. ' % (nr + 1), rozdzial, re.M)
    return rozdzial[m.end():n.start() if n else len(rozdzial)].strip('\n'), m.group(0)[len('### 18.%d. ' % nr):]


def podsekcje(tekst):
    """Tekst przed pierwszym '#### ' oraz słownik tytuł -> treść podsekcji."""
    czesci = re.split(r'^#### (.+)$', tekst, flags=re.M)
    return czesci[0].strip('\n'), {czesci[i].strip(): czesci[i + 1].strip('\n') for i in range(1, len(czesci), 2)}


def rozbij(l):
    return [c.replace('\x00', '|').strip() for c in l.replace('\\|', '\x00').strip().strip('|').split('|')]


def tabele(tekst):
    out, linie, i = [], tekst.split('\n'), 0
    while i < len(linie):
        if linie[i].startswith('|') and i + 1 < len(linie) and re.match(r'^\|[\s:|-]+\|\s*$', linie[i + 1]) and '-' in linie[i + 1]:
            nag = rozbij(linie[i])
            wyr = ['num' if c.strip().endswith(':') and not c.strip().startswith(':') else 'mid' if c.strip().startswith(':') and c.strip().endswith(':') else '' for c in rozbij(linie[i + 1])]
            wiersze = []
            i += 2
            while i < len(linie) and linie[i].startswith('|'):
                wiersze.append(rozbij(linie[i]))
                i += 1
            out.append((nag, wyr, wiersze))
        else:
            i += 1
    return out


def tabela_html(nag, wyr, wiersze, klasy_wierszy=None, komorka=None, klasa=''):
    h = [f'<div class="tw {klasa}"><table><thead><tr>' if klasa else '<div class="tw"><table><thead><tr>']
    for k, w in zip(nag, wyr):
        h.append(f'<th class="{w}">{inline(k)}</th>' if w else f'<th>{inline(k)}</th>')
    h.append('</tr></thead><tbody>')
    for wiersz in wiersze:
        kl = klasy_wierszy(wiersz) if klasy_wierszy else ''
        h.append(f'<tr class="{kl}">' if kl else '<tr>')
        for c, (txt, w) in enumerate(zip(wiersz, wyr)):
            tresc = komorka(c, txt, wiersz) if komorka else None
            tresc = inline(txt) if tresc is None else tresc
            h.append(f'<td class="{w}">{tresc}</td>' if w else f'<td>{tresc}</td>')
        h.append('</tr>')
    h.append('</tbody></table></div>')
    return ''.join(h)


def md(tekst, tabela=None):
    """Akapity, listy, tabele i nagłówki #### w kolejności z tekstu."""
    bloki, biez, typ = [], [], None
    linie = tekst.split('\n')

    def zamknij():
        nonlocal biez, typ
        if biez:
            if typ in ('ul', 'ol'):
                bloki.append(f'<{typ} class="mdlist">' + ''.join(f'<li>{inline(x)}</li>' for x in biez) + f'</{typ}>')
            else:
                bloki.append('<p>' + inline(' '.join(biez)) + '</p>')
        biez, typ = [], None

    i = 0
    while i < len(linie):
        l = linie[i]
        if l.startswith('|'):
            zamknij()
            j = i
            while j < len(linie) and linie[j].startswith('|'):
                j += 1
            for t in tabele('\n'.join(linie[i:j])):
                bloki.append(tabela(t) if tabela else tabela_html(*t))
            i = j
            continue
        if l.startswith('#### '):
            zamknij()
            bloki.append(f'<h3 class="sub3">{inline(l[5:])}</h3>')
        elif not l.strip():
            zamknij()
        elif re.match(r'^- ', l):
            if typ != 'ul':
                zamknij()
                typ = 'ul'
            biez.append(l[2:])
        elif re.match(r'^\d+\. ', l):
            if typ != 'ol':
                zamknij()
                typ = 'ol'
            biez.append(re.sub(r'^\d+\. ', '', l))
        elif typ in ('ul', 'ol') and l.startswith('  '):
            biez[-1] += ' ' + l.strip()
        else:
            if typ in ('ul', 'ol'):
                zamknij()
            typ = 'p'
            biez.append(l)
        i += 1
    zamknij()
    return '<div class="prose">' + '\n'.join(bloki) + '</div>'


def akapity(tekst):
    out, biez = [], []
    for l in tekst.split('\n') + ['']:
        if not l.strip() or l.startswith('|') or l.startswith('#') or re.match(r'^(- |\d+\. )', l):
            if biez:
                out.append(' '.join(biez))
            biez = []
            continue
        biez.append(l)
    return out


def punkty(tekst, wzor=r'^(?:- |\d+\. )(.*)$'):
    out = []
    for l in tekst.split('\n'):
        m = re.match(wzor, l)
        if m:
            out.append(m.group(1))
        elif out and l.startswith('  ') and l.strip():
            out[-1] += ' ' + l.strip()
    return out
