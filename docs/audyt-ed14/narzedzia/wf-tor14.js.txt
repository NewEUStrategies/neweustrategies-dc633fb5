const TORY = args.tory
log(`Grupa ${args.grupa}: tory ${TORY.join(', ')}`)
const wyniki = await parallel(TORY.map((tor) => () => {
  const t = { tor, klucz: tor === '3a' || tor === '3b' ? '3' : tor.startsWith('22') ? '22' : tor }
  if (tor === 'DB' || tor === 'CI') {
    return agent(tor === 'DB' ? dbPrompt(t) : ciPrompt(t), { label: `przekroj:${tor}`, phase: 'Mapa', schema: FULL_SCHEMA, effort: 'high' })
      .then((m) => ({ tor, przekroj: m }))
  }
  return parallel([
    () => agent(mapaPrompt(t), { label: `mapa:${tor}`, phase: 'Mapa', schema: MAPA_SCHEMA, effort: 'high' }),
    () => agent(defPrompt(t), { label: `defekty:${tor}`, phase: 'Defekty', schema: DEF_SCHEMA, effort: 'high' }),
  ]).then(([m, d]) => ({ tor, mapa: m, def: d }))
}))
const ok = wyniki.filter(Boolean)
return { grupa: args.grupa, wyniki: ok, brakMapy: ok.filter((w) => !w.mapa && !w.przekroj).map((w) => w.tor), brakDefektow: ok.filter((w) => !w.def && !w.przekroj).map((w) => w.tor) }
