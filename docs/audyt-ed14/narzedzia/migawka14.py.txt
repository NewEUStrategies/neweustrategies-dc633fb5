# Migawka wydania 14: wyniki torow z dziennikow przeplywow (po wznowieniu) w ksztalcie {grupa: {wyniki: [{tor, mapa, def} | {tor, przekroj}]}}
# oraz SI i SYNC jako osobne klucze. Wejscie: czesciowe14.json z zbierz14.py.
import json, sys
S = '/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed14'
d = json.load(open(S + '/czesciowe14.json'))
out, brak = {}, []
for g, w in d.items():
    if not g.startswith('T'):
        out[g] = w
        continue
    tory = sorted({k.split(':', 1)[1] for k in w})
    wyniki = []
    for t in tory:
        if f'przekroj:{t}' in w:
            wyniki.append({'tor': t, 'przekroj': w[f'przekroj:{t}']})
            continue
        m, df = w.get(f'mapa:{t}'), w.get(f'defekty:{t}')
        if m is None: brak.append(f'mapa:{t}')
        if df is None: brak.append(f'defekty:{t}')
        wyniki.append({'tor': t, 'mapa': m, 'def': df})
    out[g] = {'grupa': g, 'wyniki': wyniki}
json.dump(out, open(S + '/migawka14.json', 'w'), ensure_ascii=False, indent=1)
print('tory:', sum(len(v['wyniki']) for k, v in out.items() if k.startswith('T')), '| brak:', brak)
