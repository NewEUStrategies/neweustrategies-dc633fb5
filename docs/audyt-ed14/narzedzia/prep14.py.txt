OLD, NEW = '5b8c77c01', '346012635'

K = json.load(open(S14 + '/klasy14.json'))

    parts = l.split('\t'); st[parts[-1]] = parts[0][0]
SUB = {'3': ['3a', '3b'], '22': ['22a', '22b', '22c', '22d']}
sub_of = {}
for k, subs in SUB.items():
    for s in subs:
        for x in json.load(open(f'{S13}/wejscie/modul-{s}.json'))['plikiProdukcyjne']: sub_of[x['f']] = s
def tor_of(f, k):
    if k not in SUB: return k
    if f in sub_of: return sub_of[f]
    best, bl = SUB[k][0], -1
    for g, s in sub_of.items():
        if (now.get(g) or {}).get('k') != k and (old.get(g) or {}).get('k') != k: continue
        cp = len(os.path.commonprefix([f, g]))
        if cp > bl: best, bl = s, cp
    return best
TORY = ['1', '2', '3a', '3b'] + [str(i) for i in range(4, 22)] + ['22a', '22b', '22c', '22d', 'X-admin-shell', 'X-design-system', 'X-i18n']
mod_of_tor = lambda t: '3' if t in ('3a', '3b') else ('22' if t.startswith('22') else t)
prod = {t: [] for t in TORY}; tests = {}
for f, v in now.items():
    if v['k'] == 'NULL': continue
    if v['t']: tests.setdefault(v['k'], []).append(f); continue
    t = tor_of(f, v['k'])
    if t in prod: prod[t].append({'f': f, 'loc': v['loc']})

def klucz_pliku(p, modul):
    if p.startswith('supabase/') or p.startswith('drizzle/'): return 'DB'
    if p.startswith('.github/') or p.startswith('scripts/') or p.startswith('governance/') or re.match(r'^(vitest|playwright)[^/]*\.ts$', p) or p in ('package.json', 'bunfig.toml'): return 'CI'
    v = now.get(p) or old.get(p)
    if v and v['k'] != 'NULL': return tor_of(p, v['k'])
    m = str(modul)
    if m in SUB: return SUB[m][0]
    return m if m in TORY else 'CI'
def stan(p):
    if not p: return 'nieznany'
    if not os.path.exists(os.path.join(ROOT, p)): return 'plik_usuniety'
    if p.startswith('src/'): return 'plik_zmieniony' if p in st else 'plik_bez_zmian'
    return 'plik_zmieniony' if git('diff', '--name-only', OLD, NEW, '--', p).strip() else 'plik_bez_zmian'

                'gdzie': f"{plik}:{d.get('linia')}", 'plik': plik, 'modul': modul, 'stan13': 'nowy w ed13'})

for x in rej:

print('rejestr', len(rej), collections.Counter(x['zrodlo'] for x in rej), collections.Counter(x['stanPliku'] for x in rej))
print('tory rejestru', collections.Counter(x['tor'] for x in rej).most_common())
