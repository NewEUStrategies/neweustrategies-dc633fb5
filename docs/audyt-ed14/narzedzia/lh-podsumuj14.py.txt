import json, glob, os, sys
S = '/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed14/lh'
AUD = ['first-contentful-paint', 'largest-contentful-paint', 'total-blocking-time', 'cumulative-layout-shift', 'speed-index', 'interactive', 'server-response-time']
INS = ['render-blocking-insight', 'lcp-breakdown-insight', 'image-delivery-insight', 'cache-insight', 'cls-culprits-insight', 'document-latency-insight', 'unused-javascript', 'unsized-images', 'long-tasks', 'bootup-time', 'mainthread-work-breakdown', 'network-dependency-tree-insight', 'lcp-discovery-insight']
def opis(p):
    r = json.load(open(p))
    a = r['audits']
    w = {'plik': os.path.relpath(p, S), 'lighthouse': r.get('lighthouseVersion'), 'url': r.get('finalDisplayedUrl') or r.get('finalUrl'), 'czas': r.get('fetchTime'),
         'formFactor': r['configSettings'].get('formFactor'), 'throttling': r['configSettings'].get('throttlingMethod'),
         'kategorie': {k: round(v['score'] * 100) if v.get('score') is not None else None for k, v in r['categories'].items()},
         'metryki': {k: {'ms': round(a[k]['numericValue'], 1) if a.get(k, {}).get('numericValue') is not None else None, 'wynik': a.get(k, {}).get('score'), 'tekst': a.get(k, {}).get('displayValue')} for k in AUD},
         'statystyki': {}}
    for k in INS:
        if k in a:
            x = a[k]
            d = x.get('details') or {}
            w['statystyki'][k] = {'wynik': x.get('score'), 'tekst': x.get('displayValue'), 'oszczednoscMs': (x.get('metricSavings') or {}), 'pozycje': len(d.get('items') or [])}
    return w

    'mobile': {'urzadzenie': 'Moto G Power, 4G', 'wydajnosc': 98, 'ulatwienia': 95, 'sprawdzoneMetody': 100, 'seo': 100, 'przegladanieAgentowe': '3/3', 'FCP': '1,7 s', 'LCP': '2,1 s', 'TBT': '0 ms', 'CLS': '0', 'SI': '3,0 s',

    'przebiegi': [opis(p) for p in pliki]}
