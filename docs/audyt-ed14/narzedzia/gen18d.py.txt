# --- z map modulow ---
mech, pow_ = [], []
for g, v in M.items():
    if not g.startswith('T'):
        continue
    for x in v['wyniki']:
        mp = x.get('mapa') or {}
        for s in mp.get('synchronizacja') or []:
            mech.append({**s, 'tor': x['tor']})
        for p in mp.get('powiazania') or []:
            pow_.append({**p, 'tor': x['tor']})
def kat(s):
    t = (s.get('mechanizm', '') + ' ' + s.get('opis', '')).lower()
    if 'pas' in t and 'migrac' in t or 'drizzle' in t: return 'dwa pasy migracji'
    if 'lovable' in t or 'edytor' in t and 'publik' in t: return 'publikacja z edytora Lovable'
    if 'i18n' in t or 'slownik' in t or 'słownik' in t: return 'słowniki i18n'
    if 'skrypt' in t or 'backfill' in t or 'reczn' in t or 'ręczn' in t: return 'skrypty i akcje ręczne'
    if 'cache' in t or 'uniewazn' in t or 'unieważn' in t: return 'cache i unieważnianie'
    if 'cron' in t or 'kolejk' in t or 'harmonogram' in t or 'tick' in t: return 'harmonogram i kolejki'
    if 'typ' in t and ('generat' in t or 'types.ts' in t): return 'generowane typy i pliki'
    return 'inne'
reczne = [s for s in mech if s.get('automatyczny') is False]
kc = collections.Counter(kat(s) for s in reczne)
mc = collections.Counter(p.get('mechanizm') for p in pow_)
