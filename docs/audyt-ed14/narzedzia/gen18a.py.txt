def pl(n, miejsca=None):
    s = f'{n:.{miejsca}f}' if miejsca is not None else str(n)
    cz, _, ul = s.partition('.')
    znak = '-' if cz.startswith('-') else ''
    cz = cz.lstrip('-')
    grupy = []
    while len(cz) > 3:
        grupy.insert(0, cz[-3:])
        cz = cz[:-3]
    grupy.insert(0, cz)
    wynik = znak + ' '.join(grupy)
    return wynik + (',' + ul if ul else '')

def delta(a, b, miejsca=None):
    d = b - a
    return ('+' if d > 0 else '') + pl(d, miejsca) if d else '0'
