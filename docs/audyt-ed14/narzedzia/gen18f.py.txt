# Rozdzial 18, czesc F: krytyk kompletnosci bez obalania (przeplyw wf_29933c6d-fee) jako podsekcja 18.14.
# Wejscie: krytyk14.json (wynik przeplywu: lista spojrzen z lukami); wyjscie: r18f.md.
import json
S = '/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed14'
K = json.load(open(f'{S}/krytyk14.json', encoding='utf-8'))['krytyk']
NAZWY = {'moduly': 'moduły, funkcje i funkcjonalności', 'synchronizacja': 'synchronizacja poza 18.8', 'wdrozenie': 'CI, wdrożenie i kroki ręczne',
         'wydajnosc': 'wydajność i Speed Index', 'przekroje': 'i18n, atomic design, tenant_id, typy'}
WAGA = {'wysoka': 0, 'srednia': 1, 'niska': 2}
WAGA_PL = {'wysoka': 'wysoka', 'srednia': 'średnia', 'niska': 'niska'}
SPRAWDZONE = json.load(open(f'{S}/krytyk14-sprawdzone.json', encoding='utf-8')) if __import__('os').path.exists(f'{S}/krytyk14-sprawdzone.json') else {}
o = []
w = o.append
luki = [(r['spojrzenie'], l) for r in K for l in r['luki']]
wys = sum(1 for _, l in luki if l['waga'] == 'wysoka')
w('#### Krytyk kompletności: czego ten rozdział nie pokrył')
w('')
w(f'Po zamknięciu rozdziału {len(K)} niezależnych spojrzeń ({", ".join(NAZWY[r["spojrzenie"]] for r in K)}) szukało tego, czego rozdział nie pokrywa wobec wymagań zespołu - bez obalania ustaleń. '
  f'Zgłosiły **{len(luki)} luk**, w tym {wys} o wadze wysokiej. Każda ma dowód w repozytorium na `346012635` albo w danych wydania. '
  + ('Luki o wadze wysokiej przeczytałem w kodzie sam (oznaczone **sprawdzone**); pozostałe są zgłoszeniami do potwierdzenia przy pracy nad planem 18.13.' if SPRAWDZONE else 'Nie sprawdzałem ich osobno: to zgłoszenia do potwierdzenia przy pracy nad planem 18.13.'))
w('')
for sp, l in sorted(luki, key=lambda x: (WAGA[x[1]['waga']], list(NAZWY).index(x[0]))):
    stan = SPRAWDZONE.get(l['tytul'])
    znak = f', **{stan}**' if stan else ''
    w(f'- **({WAGA_PL[l["waga"]]}, {NAZWY[sp]}{znak}) {l["tytul"].rstrip(".")}.** {l["coBrakuje"].strip()} Dowód: {l["dowod"].strip()} '
      f'Rekomendacja: {l["rekomendacja"].strip()} Miejsce w rozdziale: {l["gdzieWRozdziale"].strip()}.')
tekst = '\n'.join(o) + '\n'
for zly in ('—', '–'):
    tekst = tekst.replace(zly, '-')
open(f'{S}/r18f.md', 'w', encoding='utf-8').write(tekst)
print('ok', len(luki), 'luk; wysokich', wys)
