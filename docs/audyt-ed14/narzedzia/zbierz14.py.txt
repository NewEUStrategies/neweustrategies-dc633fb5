# Zbiera wyniki agentow z dziennikow przeplywow wydania 14 (takze czesciowe, w trakcie przebiegu).
import json, os, sys
W = '/root/.claude/projects/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/subagents/workflows'
RUNY = {'T1': 'wf_fa536f76-af4', 'T2': 'wf_3532cdb9-d5f', 'T3': 'wf_e7489a93-632', 'T4': 'wf_94d7dd13-537', 'T5': 'wf_b25c7fec-44a',
        'T6': 'wf_70ee343b-ed7', 'T7': 'wf_93be63dd-0e9', 'T8': 'wf_27e1cc2f-35b', 'T9': 'wf_819aacd4-f52', 'SI': 'wf_fc6f7a2b-472', 'SYNC': 'wf_45384e3f-ff5'}
out, stan = {}, {}
for g, r in RUNY.items():
    etyk, wyn, uruch = {}, {}, set()
    for l in open(f'{W}/{r}/journal.jsonl'):
        e = json.loads(l)
        if e.get('type') == 'started': etyk[e['key']] = e.get('label'); uruch.add(e.get('label'))
        elif e.get('type') == 'result' and e.get('key') in etyk: wyn.setdefault(etyk[e['key']], e.get('result'))
    out[g] = wyn
    stan[g] = {'gotowe': sorted(wyn), 'wTrakcie': sorted(uruch - set(wyn))}
json.dump(out, open(sys.argv[1], 'w'), ensure_ascii=False, indent=1)
for g, s in stan.items(): print(g, 'gotowe:', s['gotowe'], '| w trakcie:', s['wTrakcie'])
