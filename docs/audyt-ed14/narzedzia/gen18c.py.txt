fmt = lambda x, n=1: f'{x:.{n}f}'.replace('.', ',')

def mediany(profil):
    pliki = sorted(glob.glob(f'{S}/lh/raporty/mediana-glowna-pl-{profil}-*.json'))
    pliki = [p for p in pliki if not p.endswith('.blad.txt')]
    if not pliki:
        return None
    zbior = {k: [] for k in ('perf', 'fcp', 'lcp', 'tbt', 'cls', 'si', 'ttfb')}
    for p in pliki:
        j = json.load(open(p))
        a = j['audits']
        zbior['perf'].append(round(j['categories']['performance']['score'] * 100))
        for k, aid in (('fcp', 'first-contentful-paint'), ('lcp', 'largest-contentful-paint'), ('tbt', 'total-blocking-time'),
                       ('cls', 'cumulative-layout-shift'), ('si', 'speed-index'), ('ttfb', 'server-response-time')):
            zbior[k].append(a[aid]['numericValue'])
    return {k: statistics.median(v) for k, v in zbior.items()} | {'n': len(pliki), 'siMin': min(zbior['si']), 'siMax': max(zbior['si'])}
