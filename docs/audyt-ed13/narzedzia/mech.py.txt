def curl(url, head=False, out=None):
    cmd=['curl','-sS','-L','-m','40','--compressed','-A',UA,'-D','-']
    cmd+=['-o', out or '/dev/null']
    cmd.append(url)
    r=subprocess.run(cmd,capture_output=True,text=True,errors='replace')
    blocks=[b for b in re.split(r'\r?\n\r?\n', r.stdout.strip()) if b.startswith('HTTP/')]
    last=blocks[-1] if blocks else ''
    status=int(re.match(r'HTTP/[\d.]+ (\d+)',last).group(1)) if last else 0
    h={}
    for l in last.splitlines()[1:]:
        if ':' in l:
            k,v=l.split(':',1); h.setdefault(k.strip().lower(),[]).append(v.strip())
    return status,h,r.stderr.strip()

def analiza(html, base):
    low=html.lower()
    def meta(name):
        m=re.search(r'<meta[^>]+(?:name|property)=["\']'+re.escape(name)+r'["\'][^>]*content=["\']([^"\']*)', html, re.I) or re.search(r'<meta[^>]+content=["\']([^"\']*)["\'][^>]*(?:name|property)=["\']'+re.escape(name)+r'["\']', html, re.I)
        return m.group(1) if m else None
    lang=re.search(r'<html[^>]*\slang=["\']([^"\']+)',html,re.I)
    canon=re.search(r'<link[^>]+rel=["\']canonical["\'][^>]*>',html,re.I)
    hreflang=set(m.group(1).lower() for m in re.finditer(r'<link[^>]+hreflang=["\']([^"\']+)',html,re.I))
    ld=[]
    for m in re.finditer(r'<script[^>]+application/ld\+json[^>]*>(.*?)</script>',html,re.I|re.S):
        try:
            j=json.loads(m.group(1).strip())
            def types(o):
                if isinstance(o,dict):
                    t=o.get('@type'); 
                    if t: ld.extend(t if isinstance(t,list) else [t])
                    for v in o.values(): types(v)
                elif isinstance(o,list):
                    for v in o: types(v)
            types(j)
        except Exception: ld.append('(niepoprawny JSON-LD)')
    rss=bool(re.search(r'application/(rss|atom)\+xml',html,re.I))
    host=urllib.parse.urlparse(base).hostname or ''
    root='.'.join(host.split('.')[-2:])
    srcs=[m.group(1) for m in re.finditer(r'<script[^>]+src=["\']([^"\']+)',html,re.I)]
    third=sorted(set(urllib.parse.urlparse(urllib.parse.urljoin(base,s)).hostname for s in srcs if urllib.parse.urlparse(urllib.parse.urljoin(base,s)).hostname and not urllib.parse.urlparse(urllib.parse.urljoin(base,s)).hostname.endswith(root)))
    return {'htmlBajty':len(html.encode()),'lang':lang.group(1) if lang else None,'title':bool(re.search(r'<title>[^<]+</title>',html,re.I)),'metaDescription':bool(meta('description')),
