# Tabela rynkowa wydania 13: K1-K11 (ocena agenta, poprawiona przez obalajacego tam, gdzie obalanie sie odbylo)
# + K12-K15 (mechanicznie: Lighthouse i naglowki). Wynik: rynek/tabela13.json.
import json
S='/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed13'
m=json.load(open(S+'/migawka.json'))
st={x['klucz']:x for x in json.load(open(S+'/rynek/strony.json'))}
mech=json.load(open(S+'/rynek/k12-15.json'))['punkty']
K=[f'K{i}' for i in range(1,12)]
wiersze=[]
for g,reg in [('NESw','NES'),('PL','PL'),('EU','UE'),('USA','USA'),('AZ','Azja')]:
    for k,v in m[g].items():
        if not k.startswith('ocena:') or not v: continue
        kl=k[6:]
        sc={o['kryterium']:o['punkty'] for o in v['oceny']}
        ob=m[g].get('obal:'+kl)
        kor=[]
        if ob:
            for kk in ob.get('korekty',[]):
                kor.append((kk['kryterium'],sc.get(kk['kryterium']),kk['powinnoByc'])); sc[kk['kryterium']]=kk['powinnoByc']
        mm=mech.get(kl,{})
        blok=('Cloudflare' if kl in ('iiss','ifri','clingendael','rsis','asan') else 'Imperva' if kl=='pism' else '')
        wiersze.append({'klucz':kl,'nazwa':st.get(kl,{}).get('nazwa',v.get('nazwa')),'region':reg,'K':sc,'suma11':sum(sc.get(x,0) for x in K),
            'zweryfikowana':bool(ob),'korekty':kor,'K12':mm.get('K12'),'K13':mm.get('K13'),'K14':mm.get('K14'),'K15':mm.get('K15'),
            'perf':mm.get('perf'),'a11y':mm.get('a11y'),'seo':mm.get('seo'),'blokada':blok,'zmierzone':mm.get('zmierzone',0),
            'wyrozniki':v.get('wyroznikiNaTleRynku'),'slabosci':v.get('slabosci'),'tech':v.get('techStack')})
json.dump(wiersze,open(S+'/rynek/tabela13.json','w'),ensure_ascii=False,indent=1)
for w in sorted(wiersze,key=lambda w:-w['suma11']):
    k15=sum(x for x in [w['K12'],w['K13'],w['K14'],w['K15']] if x is not None)
    print(f"{w['klucz']:16} {w['region']:4} {'V' if w['zweryfikowana'] else '-'} K1-11={w['suma11']:2} K12-15={k15} ({w['K12']},{w['K13']},{w['K14']},{w['K15']}) {w['blokada']} kor={w['korekty']}")
print(len(wiersze), 'zweryfikowanych', sum(w['zweryfikowana'] for w in wiersze))
