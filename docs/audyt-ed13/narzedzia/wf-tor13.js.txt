const TORY = args.tory
log(`Grupa ${args.grupa}: tory ${TORY.join(', ')}`)
const wyniki = await pipeline(
  TORY,
  (tor) => {
    const t = { tor }
    if (tor === 'DB' || tor === 'CI') {
      return agent(tor === 'DB' ? dbPrompt(t) : ciPrompt(t), { label: `przekroj:${tor}`, phase: 'Mapa', schema: FULL_SCHEMA, effort: 'high' })
        .then((m) => (m ? { tor, przekroj: m, def: { defekty: m.defekty, statusDefektow: m.statusDefektow } } : null))
    }
    return parallel([
      () => agent(mapaPrompt(t), { label: `mapa:${tor}`, phase: 'Mapa', schema: MAPA_SCHEMA, effort: 'medium' }),
      () => agent(defPrompt(t), { label: `defekty:${tor}`, phase: 'Defekty', schema: DEF_SCHEMA, effort: 'high' }),
    ]).then(([m, d]) => ({ tor, mapa: m, def: d }))
  },
  (r) => {
    if (!r) return null
    const d = r.def ?? { defekty: [], statusDefektow: [] }
    const ile = (d.defekty ?? []).length + (d.statusDefektow ?? []).length
    if (!ile) return { ...r, weryfikacja: { werdykty: [], dodatkoweZnaleziska: [] } }
    return agent(verifyPrompt({ tor: r.tor }, d), { label: `obal:${r.tor}`, phase: 'Obalanie', schema: VERIFY_SCHEMA, effort: 'high' })
      .then((v) => ({ ...r, weryfikacja: v }))
  },
)
const ok = wyniki.filter(Boolean)
return { grupa: args.grupa, wyniki: ok, brakMapy: ok.filter((w) => !w.mapa && !w.przekroj).map((w) => w.tor), brakDefektow: ok.filter((w) => !w.def).map((w) => w.tor), brak: TORY.filter((t) => !ok.find((w) => w.tor === t)) }
