// Pomiar laboratoryjny Lighthouse (mobile, symulowany throttling - jak lab w PageSpeed Insights).
// Wejscie: lista {klucz, typ, url} (JSON w argv[2]); N przebiegow na adres; wynik: maly JSON per przebieg.
// Wznawialny: pomija przebiegi, ktore juz maja plik wyniku.
import fs from "node:fs";
import { execFileSync } from "node:child_process";
const SP = "/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed13/lh";
const lista = JSON.parse(fs.readFileSync(process.argv[2], "utf8"));
const N = Number(process.argv[3] ?? 3);
const proxy = process.env.HTTPS_PROXY;
const flags = `--headless=new --no-sandbox --disable-dev-shm-usage --proxy-server=${proxy}`;
const AUDYTY = ["first-contentful-paint", "largest-contentful-paint", "total-blocking-time", "cumulative-layout-shift", "speed-index", "interactive", "server-response-time", "total-byte-weight", "dom-size", "network-requests", "mainthread-work-breakdown", "bootup-time", "third-party-summary"];
for (let run = 1; run <= N; run++) {
  for (const p of lista) {
    const out = `${SP}/wyniki/${p.klucz}-${p.typ}-${run}.json`;
    if (fs.existsSync(out)) continue;
    const tmp = `${SP}/wyniki/.tmp-${p.klucz}-${p.typ}.json`;
    const t0 = Date.now();
    let blad = null;
    try {
      execFileSync(`${SP}/node_modules/.bin/lighthouse`, [p.url, "--quiet", "--output=json", `--output-path=${tmp}`, `--chrome-flags=${flags}`, "--only-categories=performance,accessibility,best-practices,seo", "--max-wait-for-load=60000", ...(p.extra ?? [])], { env: { ...process.env, CHROME_PATH: "/opt/pw-browsers/chromium" }, stdio: ["ignore", "ignore", "pipe"], timeout: 240000 });
    } catch (e) { blad = String(e.stderr ?? e.message).slice(-600); }
    let r = { klucz: p.klucz, typ: p.typ, url: p.url, run, ms: Date.now() - t0, blad };
    if (fs.existsSync(tmp)) {
      const j = JSON.parse(fs.readFileSync(tmp, "utf8"));
      const a = j.audits ?? {};
      r = { ...r, finalUrl: j.finalDisplayedUrl ?? j.finalUrl, fetchTime: j.fetchTime, lhVersion: j.lighthouseVersion, benchmarkIndex: j.environment?.benchmarkIndex, runtimeError: j.runtimeError?.code ?? null, warnings: j.runWarnings ?? [],
        kategorie: Object.fromEntries(Object.entries(j.categories ?? {}).map(([k, v]) => [k, v.score])),
        metryki: Object.fromEntries(AUDYTY.filter((k) => a[k]).map((k) => [k, a[k].numericValue ?? null])),
        a11yBledy: Object.values(j.categories?.accessibility?.auditRefs ?? []).filter((x) => a[x.id] && a[x.id].score === 0 && x.weight > 0).map((x) => x.id),
        bpBledy: Object.values(j.categories?.["best-practices"]?.auditRefs ?? []).filter((x) => a[x.id] && a[x.id].score === 0 && x.weight > 0).map((x) => x.id),
        seoBledy: Object.values(j.categories?.seo?.auditRefs ?? []).filter((x) => a[x.id] && a[x.id].score === 0 && x.weight > 0).map((x) => x.id),
        lcpElement: a["largest-contentful-paint-element"]?.details?.items?.[0]?.items?.[0]?.node?.snippet?.slice(0, 200) ?? null,
        tytul: (j.audits?.["document-title"]?.score === 1) };
      fs.unlinkSync(tmp);
    }
    fs.writeFileSync(out, JSON.stringify(r));
    const k = r.kategorie ?? {};
    console.log(`${new Date().toISOString().slice(11, 19)} run${run} ${p.klucz}/${p.typ} ${Math.round(r.ms / 1000)}s perf=${k.performance ?? "-"} a11y=${k.accessibility ?? "-"} bp=${k["best-practices"] ?? "-"} seo=${k.seo ?? "-"} bi=${r.benchmarkIndex ?? "-"} ${r.runtimeError ?? ""} ${r.blad ? "BLAD" : ""}`);
  }
}
