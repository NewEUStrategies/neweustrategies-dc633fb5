function typy(base) {
  const all = walk(base, path.join(base, "src")).filter((f) => isTest(f) && /\.(ts|tsx|js|mjs)$/.test(f));

  for (const f of all) {

    const k = key(f);
    (res[k] ??= {})[typ] = (res[k][typ] ?? 0) + 1;
  }
  return res;
}
