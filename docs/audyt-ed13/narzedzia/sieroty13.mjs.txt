// Cztery formy importu: statyczny, dynamiczny import(), re-eksport i import czysto efektowy (import "x";) + require.
import fs from "node:fs";
import path from "node:path";

const isTest = (f) => /(^|\/)__tests__\/|\.test\.|\.spec\./.test(f);
const EXT = [".ts", ".tsx", ".js", ".jsx", ".mjs", ".cjs"];
const walk = (base, d, out = []) => { if (!fs.existsSync(d)) return out; for (const e of fs.readdirSync(d, { withFileTypes: true })) { const p = path.join(d, e.name); if (e.isDirectory()) { if (!/node_modules|\.git$|dist$|coverage/.test(e.name)) walk(base, p, out); } else if (EXT.includes(path.extname(e.name))) out.push(path.relative(base, p)); } return out; };
const RE = /(?:import|export)\s[^'"`;]*?\sfrom\s*["']([^"']+)["']|import\s*\(\s*["']([^"']+)["']\s*\)|^\s*import\s*["']([^"']+)["']|require\(\s*["']([^"']+)["']\s*\)|vi\.mock\(\s*["']([^"']+)["']/gm;
function sieroty(base) {
  const files = [...walk(base, path.join(base, "src")), ...walk(base, path.join(base, "scripts")), ...walk(base, path.join(base, "e2e"))];
  for (const f of fs.readdirSync(base)) if (EXT.includes(path.extname(f))) files.push(f);
  const set = new Set(files);
  const resolve = (from, spec) => {
    let p;
    if (spec.startsWith("@/")) p = "src/" + spec.slice(2);
    else if (spec.startsWith(".")) p = path.normalize(path.join(path.dirname(from), spec));
    else return null;
    if (set.has(p)) return p;
    for (const e of EXT) if (set.has(p + e)) return p + e;
    for (const e of EXT) if (set.has(p + "/index" + e)) return p + "/index" + e;
    const noext = p.replace(/\.(js|mjs)$/, ""); for (const e of EXT) if (set.has(noext + e)) return noext + e;
    return null;
  };
  const importers = new Map(); // tylko importy z plikow NIE-testowych licza sie jako "produkcyjny konsument"
  const importersAny = new Map();
  for (const f of files) {
    const t = fs.readFileSync(path.join(base, f), "utf8");
    for (const m of t.matchAll(RE)) {
      const spec = m[1] ?? m[2] ?? m[3] ?? m[4] ?? m[5]; const r = resolve(f, spec); if (!r || r === f) continue;
      (importersAny.get(r) ?? importersAny.set(r, new Set()).get(r)).add(f);
      if (!isTest(f)) (importers.get(r) ?? importers.set(r, new Set()).get(r)).add(f);
    }
  }
  const prod = files.filter((f) => f.startsWith("src/") && !isTest(f) && !/\.d\.ts$/.test(f));
  const entry = (f) => f.startsWith("src/routes/") || /^src\/(router|start|client|server|entry|ssr|routeTree\.gen|serviceWorker|sw)\b/.test(f) || /\.worker\./.test(f) || f.startsWith("src/test/");
  const orphan = prod.filter((f) => !entry(f) && !importersAny.has(f));
  const onlyTests = prod.filter((f) => !entry(f) && importersAny.has(f) && !importers.has(f));
  return { prod: prod.length, orphan, onlyTests };
}
