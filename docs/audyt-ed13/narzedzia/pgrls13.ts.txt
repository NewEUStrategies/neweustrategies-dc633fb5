import fs from "node:fs";
const ROOT = "/home/user/neweustrategies-dc633fb5";
const OLD = "/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed13/old-full";
const { analyzePgTapFile, isPgTapFileBroken } = await import(ROOT + "/src/lib/ci/pgTapPlan.ts");
const { extractLatestPolicies } = await import(ROOT + "/src/lib/ci/rlsPolicies.ts");
const { stripSqlComments } = await import(ROOT + "/scripts/lib/sqlMigrations.ts");
const out: Record<string, unknown> = {};
for (const [n, base] of [["ed12", OLD], ["teraz", ROOT]] as const) {
  const files = fs.readdirSync(base + "/supabase/tests").filter((f) => f.endsWith(".sql"));
  const an = files.map((f) => analyzePgTapFile(f, fs.readFileSync(`${base}/supabase/tests/${f}`, "utf8")));
  const mig = fs.readdirSync(base + "/supabase/migrations").filter((f) => f.endsWith(".sql")).sort()
    .map((file) => ({ file, sql: stripSqlComments(fs.readFileSync(`${base}/supabase/migrations/${file}`, "utf8")) }));
  const pol = extractLatestPolicies(mig);
  out[n] = { pgtapFiles: files.length, asercje: an.reduce((s, a) => s + a.counted, 0), planned: an.reduce((s, a) => s + (a.planned ?? 0), 0),
    zepsute: an.filter(isPgTapFileBroken).map((a) => a.file), polityki: pol.size, tabeleZPolitykami: new Set([...pol.values()].map((p) => p.table)).size,
    noweTesty: [] as string[] };
}
const o = out as Record<string, { pgtapFiles: number; noweTesty: string[] }>;
o.teraz.noweTesty = fs.readdirSync(ROOT + "/supabase/tests").filter((f) => f.endsWith(".sql") && !fs.existsSync(OLD + "/supabase/tests/" + f));
fs.writeFileSync("/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed13/pgrls13.json", JSON.stringify(out, null, 1));
console.log(JSON.stringify(out, null, 1));
