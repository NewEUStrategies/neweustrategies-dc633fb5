# Montaz wydania 13 w dokumencie audytu (galaz w worktree): nowy H1, naglowek wydania 13 na gorze, naglowek wydania 12
# przeniesiony pod "## Wydanie 12 (2026-10-01) - naglowek poprzedniego wydania", rozdzial 17 na koncu (sekcje 17.1-17.17
# w kolejnosci numerow, z czesci r17*.md). Idempotentny: jesli rozdzial 17 juz jest, zastepuje go.
import re, sys

S = '/tmp/claude-0/-home-user-neweustrategies-dc633fb5/208e1cdf-815a-58dd-a5d7-27f309cbff9b/scratchpad/ed13'
DOC = S + '/galaz/docs/AUDYT_POKRYCIA_TESTAMI_MODULY_FUNKCJE_2026-08-18.md'
t = open(DOC).read()
czytaj = lambda n: open(f'{S}/{n}').read().strip('\n')

# --- rozdzial 17 z czesci ---
sekcje = {}
naglowek17 = None
for czesc in ['r17e1.md', 'r17a.md', 'r17b.md', 'r17d.md', 'r17c.md', 'r17e2.md']:
    biezaca = None
    for linia in czytaj(czesc).split('\n'):
        if linia.startswith('## 17. '):
            naglowek17 = [linia]
            biezaca = naglowek17
            continue
        m = re.match(r'^### 17\.(\d+)\. ', linia)
        if m:
            k = int(m.group(1))
            assert k not in sekcje, f'sekcja 17.{k} dwa razy ({czesc})'
            sekcje[k] = [linia]
            biezaca = sekcje[k]
            continue
        assert biezaca is not None, f'{czesc}: tekst przed pierwszym naglowkiem: {linia[:60]}'
        biezaca.append(linia)
assert naglowek17 and sorted(sekcje) == list(range(1, 18)), sorted(sekcje)
sekcje[8] = '\n'.join(sekcje[8]).rstrip('\n').split('\n') + [''] + czytaj('r17f.md').split('\n')
blok = lambda linie: re.sub(r'\n{3,}', '\n\n', '\n'.join(linie).strip('\n'))
rozdzial = '\n\n'.join([blok(naglowek17)] + [blok(sekcje[k]) for k in range(1, 18)])

# --- naglowek dokumentu ---
H1_STARY = '# Audyt pokrycia testami: moduł po module, funkcja po funkcji (2026-10-01)'
H1_NOWY = '# Audyt pokrycia testami: moduł po module, funkcja po funkcji (2026-10-04)'
W12 = '## Wydanie 12 (2026-10-01) - nagłówek poprzedniego wydania'
W11 = '## Wydanie 11 (2026-09-12) - nagłówek poprzedniego wydania'
if t.startswith(H1_STARY):
    assert W12 not in t
    i11 = t.index('\n---\n\n' + W11)
    stary12 = t[len(H1_STARY):i11].strip('\n')
    assert stary12.startswith('**WYDANIE 12'), stary12[:40]
    t = (H1_NOWY + '\n\n' + czytaj('naglowek13.md') + '\n\n---\n\n' + W12 + '\n\n' + stary12 + '\n' + t[i11:])
else:
    # ponowny montaz: podmien tylko naglowek wydania 13 (miedzy H1 a "---" przed W12)
    assert t.startswith(H1_NOWY)
    i12 = t.index('\n---\n\n' + W12)
    t = H1_NOWY + '\n\n' + czytaj('naglowek13.md') + '\n' + t[i12:]

# --- rozdzial 17 na koncu (albo zamiana istniejacego) ---
i17 = t.find('\n## 17. ')
if i17 >= 0:
    t = t[:i17].rstrip('\n') + '\n'
t = t.rstrip('\n') + '\n\n' + rozdzial + '\n'
for zly in ('\u2014', '\u2013'):
    assert zly not in rozdzial, f'znak {zly!r} w rozdziale 17'
open(DOC, 'w').write(t)
print('ok', len(t.split('\n')), 'linii; rozdzial 17:', len(rozdzial.split('\n')), 'linii')
