# ---------- markdown (podzbior uzyty w rozdziale 17) -> HTML ----------
def inline(t):
    t = html.escape(t.strip(), quote=False)
    czesci = re.split(r'(`[^`]+`)', t)
    out = []
    for c in czesci:
        if c.startswith('`') and c.endswith('`') and len(c) > 1:
            out.append('<code>' + c[1:-1] + '</code>')
        else:
            c = re.sub(r'\*\*(.+?)\*\*', r'<b>\1</b>', c)
            out.append(c)
    return ''.join(out)


def sekcja(nr):
    m = re.search(r'^### 17\.%d\. .*$' % nr, rozdzial, re.M)
    n = re.search(r'^### 17\.%d\. ' % (nr + 1), rozdzial, re.M)
    return rozdzial[m.end():n.start() if n else len(rozdzial)].strip('\n'), m.group(0)[len('### 17.%d. ' % nr):]


def tabele(tekst):
    """Lista tabel markdown: (naglowki, wyrownania, wiersze) - komorki surowe (markdown)."""
    out = []
    linie = tekst.split('\n')
    i = 0
    while i < len(linie):
        if linie[i].startswith('|') and i + 1 < len(linie) and re.match(r'^\|[\s:|-]+\|\s*$', linie[i + 1]) and '-' in linie[i + 1]:
            rozbij = lambda l: [c.replace('\x00', '|').strip() for c in l.replace('\\|', '\x00').strip().strip('|').split('|')]
            nag = rozbij(linie[i])
            wyr = ['num' if c.strip().endswith(':') else '' for c in rozbij(linie[i + 1])]
            wiersze = []
            i += 2
            while i < len(linie) and linie[i].startswith('|'):
                wiersze.append(rozbij(linie[i]))
                i += 1
            out.append((nag, wyr, wiersze))
        else:
            i += 1
    return out


def tabela_html(nag, wyr, wiersze, klasy_wierszy=None, komorka=None):
    h = ['<div class="tw"><table><thead><tr>']
    for k, w in zip(nag, wyr):
        h.append(f'<th class="{w}">{inline(k)}</th>' if w else f'<th>{inline(k)}</th>')
    h.append('</tr></thead><tbody>')
    for r, wiersz in enumerate(wiersze):
        kl = klasy_wierszy(wiersz) if klasy_wierszy else ''
        h.append(f'<tr class="{kl}">' if kl else '<tr>')
        for c, (txt, w) in enumerate(zip(wiersz, wyr)):
            tresc = komorka(c, txt, wiersz) if komorka else None
            tresc = inline(txt) if tresc is None else tresc
            h.append(f'<td class="{w}">{tresc}</td>' if w else f'<td>{tresc}</td>')
        h.append('</tr>')
    h.append('</tbody></table></div>')
    return ''.join(h)


def proza(tekst, bez_tabel=True):
    """Akapity, listy punktowane i numerowane; tabele pomija (renderowane osobno)."""
    bloki, biez, typ = [], [], None

    def zamknij():
        nonlocal biez, typ
        if biez:
            if typ == 'ul':
                bloki.append('<ul class="mdlist">' + ''.join(f'<li>{inline(x)}</li>' for x in biez) + '</ul>')
            elif typ == 'ol':
                bloki.append('<ol class="mdlist">' + ''.join(f'<li>{inline(x)}</li>' for x in biez) + '</ol>')
            else:
                bloki.append('<p>' + inline(' '.join(biez)) + '</p>')
        biez, typ = [], None

    for l in tekst.split('\n'):
        if l.startswith('|'):
            zamknij()
            continue
        if not l.strip():
            zamknij()
            continue
        m_ul = re.match(r'^- (.*)$', l)
        m_ol = re.match(r'^\d+\. (.*)$', l)
        if m_ul:
            if typ != 'ul':
                zamknij()
                typ = 'ul'
            biez.append(m_ul.group(1))
        elif m_ol:
            if typ != 'ol':
                zamknij()
                typ = 'ol'
            biez.append(m_ol.group(1))
        elif typ in ('ul', 'ol') and l.startswith('  '):
            biez[-1] += ' ' + l.strip()
        else:
            if typ in ('ul', 'ol'):
                zamknij()
            typ = 'p'
            biez.append(l)
    zamknij()
    return '\n'.join(bloki)


def akapity(tekst):
    """Akapity jako lista surowych tekstow (markdown), bez list i tabel."""
    out, biez = [], []
    for l in tekst.split('\n') + ['']:
        if not l.strip() or l.startswith('|') or re.match(r'^(- |\d+\. )', l):
            if biez:
                out.append(' '.join(biez))
            biez = []
            continue
        biez.append(l)
    return out


def punkty(tekst):
    out = []
    for l in tekst.split('\n'):
        m = re.match(r'^- (.*)$', l)
        if m:
            out.append(m.group(1))
        elif out and l.startswith('  ') and l.strip():
            out[-1] += ' ' + l.strip()
    return out
