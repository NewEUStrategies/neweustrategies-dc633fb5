def n(x, nd=None):
    if isinstance(x, float):
        s = f'{x:,.{2 if nd is None else nd}f}'
    else:
        s = f'{x:,}'
    return s.replace(',', ' ').replace('.', ',')


def d(a, b, nd=None):
    v = b - a
    if isinstance(v, float):
        v = round(v, 2 if nd is None else nd)
    return ('+' if v > 0 else '') + n(v, nd) if v != 0 else '0'


def pl(k, a, b, c):
    k = int(k)
    if k == 1:
        return f'{k} {a}'
    if 2 <= k % 10 <= 4 and not 12 <= k % 100 <= 14:
        return f'{n(k)} {b}'
    return f'{n(k)} {c}'


def p2(x):
    return n(float(x))


def pc(a, b):
    return 100.0 * a / b if b else 100.0


def cl(s):
    return re.sub(r'\s+', ' ', str(s)).replace('|', '/').replace('\u2014', '-').replace('\u2013', '-').strip()
