const rel = (p) => p.replace(ROOT + "/", "");
const key = (f) => { const c = mm.classifyPath(f); return c.module != null ? String(c.module) : c.crossCutting ? "X-" + c.crossCutting : "NULL"; };
export function met(e) {
  const L = new Map();
  for (const [id, loc] of Object.entries(e.statementMap)) { const n = loc.start.line; L.set(n, Math.max(L.get(n) ?? 0, e.s[id] ?? 0)); }
  const fm = Object.entries(e.f), bm = Object.values(e.b).flat();
  const deadNamed = fm.filter(([id, h]) => h === 0 && e.fnMap[id] && !/^\(anonymous_\d+\)$/.test(e.fnMap[id].name)).map(([id]) => ({ name: e.fnMap[id].name, line: e.fnMap[id].loc?.start?.line ?? e.fnMap[id].line }));
  const uncoveredLines = [...L.entries()].filter(([, h]) => h === 0).map(([n]) => n).sort((a, b) => a - b);
  return { L: L.size, Lc: L.size - uncoveredLines.length, S: Object.keys(e.s).length, Sc: Object.values(e.s).filter((x) => x > 0).length, F: fm.length, Fc: fm.filter(([, h]) => h > 0).length, B: bm.length, Bc: bm.filter((x) => x > 0).length, deadNamed, uncoveredLines };
}
const perFile = {}, perKey = {};
const T = { L: 0, Lc: 0, S: 0, Sc: 0, F: 0, Fc: 0, B: 0, Bc: 0, files: 0, zero: 0, dead: 0 };
for (const [abs, e] of Object.entries(cov)) {
  const f = rel(abs); const m = met(e); const k = key(f);
  perFile[f] = { k, L: m.L, Lc: m.Lc, F: m.F, Fc: m.Fc, B: m.B, Bc: m.Bc, S: m.S, Sc: m.Sc, dead: m.deadNamed, unc: m.uncoveredLines };
  const r = (perKey[k] ??= { L: 0, Lc: 0, S: 0, Sc: 0, F: 0, Fc: 0, B: 0, Bc: 0, files: 0, zero: 0, dead: 0, zeros: [] });
  for (const q of ["L", "Lc", "S", "Sc", "F", "Fc", "B", "Bc"]) { r[q] += m[q]; T[q] += m[q]; }
  r.files++; T.files++; r.dead += m.deadNamed.length; T.dead += m.deadNamed.length;
  if (m.L > 0 && m.Lc === 0) { r.zero++; T.zero++; r.zeros.push([f, m.L]); }
}
const pc = (a, b) => (b ? (100 * a) / b : 100);
