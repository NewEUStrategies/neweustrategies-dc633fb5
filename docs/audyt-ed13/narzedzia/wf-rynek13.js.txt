function ocenPrompt(s) {
  return `Oceniasz strone internetowa osrodka analitycznego (think tanku) ${s.nazwa} (${s.kraj}), adres ${s.url}, na potrzeby porownania rynkowego platformy New European Strategies. Interesuje nas jakosc strony jako PRODUKTU cyfrowego (co czytelnik, ekspert, darczynca i dziennikarz moga na niej zrobic), nie jakosc analiz merytorycznych. ${RUBRYKA}
${ZASADY}
Klucz strony: ${s.klucz}. Zwroc oceny dla wszystkich kryteriow K1-K11.`
}
function obalPrompt(s, o) {
  const lista = (o.oceny ?? []).map((x) => `${x.kryterium}: ${x.punkty} pkt - ${String(x.obserwacje).slice(0, 400)} | dowody: ${(x.dowodyUrl ?? []).slice(0, 3).join(' ')}`).join('\n')
  return `Jestes niezaleznym weryfikatorem porownania rynkowego. Inny agent ocenil strone ${s.nazwa} (${s.url}) wedlug rubryki ponizej. Twoim zadaniem jest OBALIC jego oceny tam, gdzie sie pomylil. Sprawdz SAMODZIELNIE (pobierajac strony) co najmniej: wszystkie oceny 0 i 3, kazde twierdzenie o braku funkcji (wyszukiwarka, wersje jezykowe, newsletter, wydarzenia, konta) oraz K4 i K5 niezaleznie od oceny. ${RUBRYKA}
${ZASADY}
OCENY AGENTA:
${lista}

korekty - tylko tam, gdzie masz dowod, ze ocena jest bledna (podaj URL i obserwacje). potwierdzone - kryteria, ktore sprawdziles i sa poprawne.`
}
